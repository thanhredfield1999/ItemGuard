31:     replacements = {}
32:     with zipfile.ZipFile(source) as original:
33:         if 'org/sqlite/JDBC.class' not in original.namelist():
34:             raise RuntimeError('Not a shaded artifact')

36:             replacements[name] = original.read('lite/' + name)
