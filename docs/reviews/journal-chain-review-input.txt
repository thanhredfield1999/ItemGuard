2|    // Byte/hash/sequence layer only, not trusted genesis, schema, lifecycle or historyComplete.
3|    // Caller supplies one complete line from EACH mirror in order; never filter/reorder.
4|    public sealed class CommonJournalChain {
5|        readonly int maximumRecords;
6|        string previousLineHash = new string('0', 64);
7|        public int CommonRecordCount { get; private set; }
8|        public bool Stopped { get; private set; }
9|        public CommonJournalChain(int maximumRecords) {
10|            if (maximumRecords <= 0) throw new System.ArgumentOutOfRangeException("maximumRecords");
11|            this.maximumRecords = maximumRecords;
12|        }
13|        bool Stop() { Stopped = true; return false; }
14|        public bool AppendPair(byte[] first, byte[] second) {
15|            if (Stopped) return false;
16|            if (first == null || second == null || CommonRecordCount >= maximumRecords
17|                || first.Length == 0 || first.Length > CanonicalDecoder.MaxDocumentBytes
18|                || first.Length != second.Length) return Stop();
19|            for (int i = 0; i < first.Length; i++) if (first[i] != second[i]) return Stop();
20|            try {
21|                var record = CanonicalDecoder.VerifyJournalLine(first);
22|                if (!(record["seq"] is ulong) || (ulong)record["seq"] != (ulong)CommonRecordCount
23|                    || !(record["previousRecordSha256"] is string)
24|                    || (string)record["previousRecordSha256"] != previousLineHash) return Stop();
25|                // Hash exact complete prior line bytes, including LF; no retained array alias.
26|                using (var sha = System.Security.Cryptography.SHA256.Create()) {
27|                    var text = new System.Text.StringBuilder(64);
28|                    foreach (byte b in sha.ComputeHash(first)) text.Append(b.ToString("x2"));
29|                    previousLineHash = text.ToString();
30|                }
31|                CommonRecordCount++;
32|                return true;
33|            } catch (System.FormatException) { return Stop(); }
34|        }
35|    }

40|        /// <summary>Verifies one journal line: exactly one terminating LF, strict canonical
41|        /// body, canonical round-trip byte equality, a lowercase hex-64 <c>recordSha256</c>
42|        /// member, and a hash matching the deletion pre-image.</summary>
43|        public static IDictionary VerifyJournalLine(byte[] line) {
44|            if (line == null) throw new FormatException("line required");
45|            if (line.Length < 2) throw new FormatException("line too short");
46|            if (line[line.Length - 1] != 0x0A) throw new FormatException("line must end with exactly one LF");
47|            if (line[line.Length - 2] == 0x0A) throw new FormatException("line must end with exactly one LF");
48|            if (line[line.Length - 2] == 0x0D) throw new FormatException("CR is not permitted");
49|
50|            var body = new byte[line.Length - 1];
51|            Buffer.BlockCopy(line, 0, body, 0, body.Length);
52|            for (int k = 0; k < body.Length; k++) {
53|                if (body[k] == 0x0A) throw new FormatException("embedded LF in record body");
54|            }
55|
56|            IDictionary record = ParseDocument(body);
57|
58|            // Round-trip: rebuild canonical bytes and demand exact equality before hashing.
59|            byte[] rebuilt = CanonicalReEncoder.Encode(record);
60|            if (rebuilt.Length != body.Length) throw new FormatException("record is not exact canonical bytes");
61|            for (int k = 0; k < body.Length; k++) {
62|                if (rebuilt[k] != body[k]) throw new FormatException("record is not exact canonical bytes");
63|            }
64|
65|            object declared = record.Contains(RecordHashMember) ? record[RecordHashMember] : null;
66|            if (declared == null) throw new FormatException("record is missing " + RecordHashMember);
67|            var hex = declared as string;
68|            if (!IsLowerHex64(hex)) throw new FormatException(RecordHashMember + " must be lowercase hex-64");
69|            if (hex != ComputeRecordSha256(record)) throw new FormatException("record hash mismatch");
70|            return record;
71|        }

73|        public static bool IsLowerHex64(string value) {
74|            if (value == null || value.Length != 64) return false;
75|            foreach (char c in value) {
76|                bool ok = (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f');
77|                if (!ok) return false;
78|            }
79|            return true;
80|        }
