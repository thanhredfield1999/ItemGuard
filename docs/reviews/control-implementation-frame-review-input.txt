1|namespace ItemGuard.Execution.Control {
2|    // One length-prefixed frame, no stream/OS access and no reset/reuse.
3|    public sealed class ControlFrameReader {
4|        readonly byte[] header = new byte[4];
5|        int headerCount;
6|        byte[] body;
7|        int bodyCount;
8|        public bool Failed { get; private set; }
9|        public bool Complete { get { return !Failed && body != null && bodyCount == body.Length; } }
10|        bool Reject() { Failed = true; return false; }
11|        public bool EndOfInput() { return Complete || Reject(); }
12|        public bool Append(byte[] chunk, int offset, int count) {
13|            if (Failed) return false;
14|            if (chunk == null || offset < 0 || count < 0 || offset > chunk.Length - count) return Reject();
15|            for (int i = offset; i < offset + count; i++) {
16|                if (headerCount < 4) {
17|                    header[headerCount++] = chunk[i];
18|                    if (headerCount == 4) {
19|                        uint length = (uint)header[0] | ((uint)header[1] << 8)
20|                            | ((uint)header[2] << 16) | ((uint)header[3] << 24);
21|                        // Bound before allocation, including unsigned high-bit lengths.
22|                        if (length == 0 || length > 65536) return Reject();
23|                        body = new byte[(int)length];
24|                    }
25|                } else {
26|                    if (bodyCount == body.Length) return Reject();
27|                    body[bodyCount++] = chunk[i];
28|                }
29|            }
30|            return true;
31|        }
32|        public byte[] GetBody() {
33|            if (!Complete) throw new System.InvalidOperationException("frame is not complete and valid");
34|            return (byte[])body.Clone();
35|        }
36|    }

1|using System;
2|using ItemGuard.Execution.Control;
3|
4|class ControlFrameBoundsTests {
5|    static int cases;
6|    static int failures;
7|    static void Check(bool value, string message) { if (!value) throw new Exception(message); }
8|    static void Run(string name, Action test) {
9|        cases++;
10|        try { test(); }
11|        catch (Exception e) { failures++; Console.WriteLine("FAIL " + name + ": " + e.Message); }
12|    }
13|    static byte[] Frame(int size) {
14|        var frame = new byte[size + 4];
15|        frame[0] = (byte)size; frame[1] = (byte)(size >> 8);
16|        frame[2] = (byte)(size >> 16); frame[3] = (byte)(size >> 24);
17|        for (int i = 4; i < frame.Length; i++) frame[i] = (byte)i;
18|        return frame;
19|    }
20|    static void NoBody(ControlFrameReader reader) {
21|        bool rejected = false;
22|        try { reader.GetBody(); } catch (InvalidOperationException) { rejected = true; }
23|        Check(rejected, "partial or failed body must not escape");
24|    }
25|    static void Fragment(int size, int piece) {
26|        var frame = Frame(size);
27|        var reader = new ControlFrameReader();
28|        for (int offset = 0; offset < frame.Length; offset += piece) {
29|            int count = Math.Min(piece, frame.Length - offset);
30|            Check(reader.Append(frame, offset, count), "fragment accepted");
31|            Check(reader.Complete == (offset + count == frame.Length), "exact completion");
32|        }
33|        var body = reader.GetBody();
34|        for (int i = 0; i < body.Length; i++) Check(body[i] == frame[i + 4], "all bytes exact");
35|        Check(reader.EndOfInput(), "EOF after full frame");
36|    }
37|    static void BadLength(uint length) {
38|        var reader = new ControlFrameReader();
39|        byte[] header = { (byte)length, (byte)(length >> 8), (byte)(length >> 16), (byte)(length >> 24) };
40|        Check(!reader.Append(header, 0, 4) && reader.Failed && !reader.Complete, "length rejected before allocation");
41|        NoBody(reader);
42|        Check(!reader.Append(Frame(1), 0, 5), "cannot reuse failed reader");
43|    }
44|    static void Truncated(int prefix) {
45|        var reader = new ControlFrameReader();
46|        var frame = Frame(10);
47|        Check(reader.Append(frame, 0, prefix) && !reader.Complete, "truncated frame waits");
48|        NoBody(reader);
49|        Check(!reader.EndOfInput() && reader.Failed, "EOF rejects prefix");
50|        Check(!reader.Append(frame, prefix, frame.Length - prefix), "late suffix cannot repair");
51|    }
52|    static void Trailing(bool sameChunk) {
53|        var reader = new ControlFrameReader();
54|        var frame = Frame(3);
55|        if (sameChunk) {
56|            Array.Resize(ref frame, frame.Length + 1);
57|            Check(!reader.Append(frame, 0, frame.Length), "trailing byte in same read");
58|        } else {
59|            Check(reader.Append(frame, 0, frame.Length), "valid frame");
60|            Check(!reader.Append(new byte[] { 0 }, 0, 1), "later trailing byte");
61|        }
62|        Check(reader.Failed && !reader.Complete, "no complete body after trailing data");
63|        NoBody(reader);
64|    }
65|    static int Main() {
66|        foreach (int size in new [] { 1, 2, 255, 256, 65535, 65536 })
67|            foreach (int piece in new [] { 1, 3, 17, 4096, 70000 }) {
68|                int s = size, p = piece;
69|                Run("fragment size=" + s + " piece=" + p, delegate { Fragment(s, p); });
70|            }
71|        foreach (uint length in new uint[] { 0, 65537, 0x80000000, uint.MaxValue }) {
72|            uint n = length;
73|            Run("length=" + n, delegate { BadLength(n); });
74|        }
75|        for (int prefix = 0; prefix < 14; prefix++) {
76|            int p = prefix;
77|            Run("truncated prefix=" + p, delegate { Truncated(p); });
78|        }
79|        Run("same-read trailing byte", delegate { Trailing(true); });
80|        Run("later trailing byte", delegate { Trailing(false); });
81|        Run("invalid buffer ranges", delegate {
82|            var reader = new ControlFrameReader();
83|            Check(!reader.Append(new byte[4], int.MaxValue, 1), "range overflow reject");
84|            Check(reader.Failed, "sticky range error");
85|        });
86|        Console.WriteLine("FRAME_BOUNDS cases=" + cases + " failures=" + failures);
87|        return failures == 0 ? 0 : 1;
88|    }
