589|## Protocol/state machine
590|
591|Guardian là writer duy nhất. Mọi append đi qua một serialized writer queue;
592|allocate seq, prior-line hash, write/flush E: rồi write/flush C: thuộc cùng một
593|serialized operation. Mirror failure làm writer terminal-failed, không nhận
594|append tiếp hoặc ACK success. Chỉ protocol-speaking worker gửi request và block
595|đến ACK/NAK; silent roles không có control channel.
596|
597|### Capability transport
598|
599|- Sealed role manifest bắt buộc khai `controlProtocolPolicy=SPEAKER|SILENT`.
600|  SILENT roles không có control pipes, request reader hoặc ACK deadline. SPEAKER
601|  roles nhận hai anonymous pipes: request worker→guardian và response
602|  guardian→worker. Role không khai policy bị reject trước spawn.
603|- Guardian giữ request-read/response-write ends non-inheritable. Exact
604|  `PROC_THREAD_ATTRIBUTE_HANDLE_LIST` là union của ba child stdio ends
605|  (`stdin-read`, `stdout-write`, `stderr-write`) và, chỉ với SPEAKER, hai control
606|  ends (`request-write`, `response-read`). Mỗi listed handle inheritable;
607|  `bInheritHandles=TRUE`, `STARTF_USESTDHANDLES` và
608|  `EXTENDED_STARTUPINFO_PRESENT` bắt buộc. STARTUPINFO std handles phải khớp
609|  exact union; missing/extra handle block trước resume. Không broad inheritance.
610|  `stdinPolicy=none` vẫn cấp stdin-read, guardian đóng stdin-write để tạo EOF.
611|  Tất cả worker descendants bị cấm bởi `ActiveProcessLimit=1`.
612|- Control frames không dùng stdin/stdout/stderr. Normal process output chỉ được
613|  guardian drain vào bounded role logs; không có push/data pipe thứ hai.
614|- Frame là `uint32-le length` + exact canonical UTF-8 JSON bytes, tối đa 64 KiB.
615|  Schema cố định gồm `schemaVersion`, monotonic `requestId`, allowlisted
616|  `requestType`, `role`, `payload`. Unknown/duplicate key, length 0/oversize,
617|  partial body, trailing bytes hoặc out-of-order request bị reject.
618|- ACK/NAK frame bind exact `requestId` và exact SHA-256 của request bytes. Worker
619|  không action nếu chưa nhận ACK matching cả hai giá trị.
620|- Guardian đọc đủ request frame trước ACK, worker write đầy đủ request rồi đọc
621|  response; mỗi worker tối đa một outstanding request. Reads/writes loop trên
622|  partial transfer với deadline; không dựa pipe buffer chứa được toàn frame.
623|  Test frame 64 KiB trên buffer nhỏ phải complete, không timeout/deadlock.
624|- Mỗi request có deadline monotonic cụ thể trong role manifest; deadline tối đa
625|  30 giây cho journal ACK. Timeout, unexpected EOF, broken pipe, malformed frame, duplicate
626|  request hoặc unsolicited bytes làm guardian NAK khi có thể, ghi
627|  `PROTOCOL_REJECTED` nếu journal còn usable và đóng mọi worker jobs.
628|- SILENT role exit không thể gây control EOF failure vì không có control channel.
629|  SPEAKER clean close cần terminal-close protocol được đặc tả và test riêng trước
630|  implementation transport: sau khi hoàn tất mọi request trước, SPEAKER gửi
631|  `CONTROL_CLOSE` như request cuối. Guardian validate không pending operation,
632|  journal `<ROLE>_CONTROL_CLOSE_ACCEPTED`, rồi ACK matching request hash và ID.
633|  Worker chỉ sau ACK mới đóng request-write và exit trong deadline manifest.
634|  Sau accepted-close guardian không nhận thêm frame; unexpected bytes, partial
635|  frame, duplicate close hoặc exit deadline expiry fail-closed. EOF sau close
636|  accepted không tự là exit; vẫn phải wait/drain/seal/exit như lifecycle chung.
637|  EOF trước close accepted fail-closed dù process exit 0; không suy verdict từ
638|  scheduling order giữa EOF và wait notification. ACK write failure vẫn failure,
639|  không biến close accepted thành proof worker nhận ACK.
640|- Pipe handle là runtime capability, không phải cryptographic identity. Threat
641|  model cùng-user được giữ bằng explicit handle inheritance + guardian process
642|  deny `PROCESS_DUP_HANDLE`; named pipe/TCP/file polling không được dùng làm
643|  control channel.

1|namespace ItemGuard.Execution.Control {
2|    public enum ControlRole { SPEAKER, SILENT }
3|    public enum ControlRequestKind { OPERATION, CONTROL_CLOSE }
4|
5|    // Pure serialized policy. Inputs must already be validated by the transport,
6|    // schema/identity validator and durable writer; this class proves none of those facts.
7|    public sealed class ControlSession {
8|        enum Stage { Idle, Reading, Ready, Dispatched, Committed, Closing }
9|        readonly ControlRole role;
10|        Stage stage;
11|        bool released;
12|        ulong requestId;
13|        bool hasRequestId;
14|        string requestHash;
15|        ControlRequestKind requestKind;
16|        bool closeAckWritten;
17|        bool eofObserved;
18|        bool exitObserved;
19|        public bool Failed { get; private set; }
20|        public bool CloseAccepted { get; private set; }
21|        // Control-channel completion only, NOT lifecycle output seal or product success.
22|        public bool ClosedCleanly { get { return !Failed && CloseAccepted && closeAckWritten && eofObserved && exitObserved; } }
23|        public bool ObserveEof() {
24|            if (Failed) return false;
25|            if (!CloseAccepted || eofObserved) return Reject();
26|            eofObserved = true; return true;
27|        }
28|        public bool ObserveExit() {
29|            if (Failed) return false;
30|            if (exitObserved || (role == ControlRole.SPEAKER && !CloseAccepted)) return Reject();
31|            exitObserved = true; return true;
32|        }
33|        public bool CanAcknowledge { get { return !Failed && stage == Stage.Committed; } }
34|        public ControlSession(ControlRole role) {
35|            if (role != ControlRole.SPEAKER && role != ControlRole.SILENT)
36|                throw new System.ArgumentOutOfRangeException("role");
37|            this.role = role;
38|        }
39|        bool Reject() { Failed = true; return false; }
40|        // Transport calls this on timeout, broken pipe, invalid frame or ACK write failure.
41|        public void Fail() { Reject(); }
42|        public bool BeginFrame() {
43|            if (Failed) return false;
44|            if (role != ControlRole.SPEAKER || stage != Stage.Idle) return Reject();
45|            stage = Stage.Reading; return true;
46|        }
47|        public bool CompleteFrame(ulong id, string hash, ControlRequestKind kind) {
48|            if (Failed) return false;
49|            if (stage != Stage.Reading || !ValidHash(hash)
50|                || (kind != ControlRequestKind.OPERATION && kind != ControlRequestKind.CONTROL_CLOSE)
51|                || (hasRequestId && id <= requestId)) return Reject();
52|            hasRequestId = true;
53|            requestId = id; requestHash = hash; requestKind = kind; stage = Stage.Ready; return true;
54|        }
55|        public bool TryDispatch(bool hasPendingOperation) {
56|            if (Failed) return false;
57|            if (stage != Stage.Ready) return Reject();
58|            if (!released) return false;
59|            if (requestKind == ControlRequestKind.CONTROL_CLOSE && hasPendingOperation) return Reject();
60|            stage = Stage.Dispatched; return true;
61|        }
62|        public bool ReleaseCommitted() {
63|            if (Failed) return false;
64|            if (released) return Reject();
65|            released = true; return true;
66|        }
67|        public bool CommitRequest(bool firstMirror, bool secondMirror) {
68|            if (Failed) return false;
69|            if (stage != Stage.Dispatched || !firstMirror || !secondMirror) return Reject();
70|            stage = Stage.Committed;
71|            if (requestKind == ControlRequestKind.CONTROL_CLOSE) CloseAccepted = true;
72|            return true;
73|        }
74|        static bool ValidHash(string value) {
75|            if (value == null || value.Length != 64) return false;
76|            foreach (char c in value)
77|                if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f'))) return false;
78|            return true;
79|        }
80|        public bool AcknowledgeWritten(ulong id, string hash) {
81|            if (Failed) return false;
82|            if (!CanAcknowledge || id != requestId || hash != requestHash) return Reject();
83|            if (CloseAccepted) { closeAckWritten = true; stage = Stage.Closing; }
84|            else stage = Stage.Idle;
85|            return true;
86|        }
87|    }
