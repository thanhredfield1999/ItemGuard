FILE tools/execution-authority/lifecycle/PaperLifecycle.cs
1|namespace ItemGuard.Execution {
2|    // Consumes only externally validated common-prefix events for one process identity.
3|    public sealed class PaperLifecycle {
4|        public string Process { get; private set; }
5|        public string Code { get; private set; }
6|        public string Output { get; private set; }
7|        public string Exit { get; private set; }
8|        public bool InvalidSuffix { get; private set; }
9|        public PaperLifecycle() { Process = "ABSENT"; Code = Output = Exit = "NOT_APPLICABLE"; }
10|        public bool Apply(string kind) {
11|            if (InvalidSuffix) return false;
12|            if (kind == "SPAWN_INTENT" && Process == "ABSENT") { Process = "SPAWN_INTENT_ONLY"; return true; }
13|            if (kind == "PROCESS_CREATED_SUSPENDED" && Process == "SPAWN_INTENT_ONLY") {
14|                Process = "CREATED_SUSPENDED"; Code = Output = Exit = "NOT_ESTABLISHED"; return true;
15|            }
16|            if (kind == "EXECUTION_RELEASE_INTENT" && Process == "CREATED_SUSPENDED") { Process = "RELEASE_INTENT_UNCERTAIN"; return true; }
17|            if (kind == "EXECUTION_RELEASE_OBSERVED" && Process == "RELEASE_INTENT_UNCERTAIN") { Process = "RELEASED"; return true; }
18|            if (kind == "SUSPENDED_TERMINATION_INTENT" && Process == "CREATED_SUSPENDED") { Process = "SUSPENDED_TERMINATION_INTENT_UNCERTAIN"; return true; }
19|            if (kind == "SUSPENDED_TERMINATION_OBSERVED" && Process == "SUSPENDED_TERMINATION_INTENT_UNCERTAIN") { Process = "TERMINATED_SUSPENDED"; Code = "NON_EXECUTION_ESTABLISHED"; return true; }
20|            if (kind == "EXECUTION_ACTIVITY_OBSERVED" && Process == "RELEASED" && Code == "NOT_ESTABLISHED" && Output == "NOT_ESTABLISHED") { Code = "ESTABLISHED"; return true; }
21|            // Caller validates the seal's factual counters. A released process with no
22|            // activity evidence may seal, but its code execution remains uncertain.
23|            if (kind == "OUTPUT_SEALED" && Output == "NOT_ESTABLISHED" && (Process == "TERMINATED_SUSPENDED" || Process == "RELEASED")) { Output = "ESTABLISHED"; return true; }
24|            if (kind == "EXIT_OBSERVED" && Output == "ESTABLISHED" && Exit == "NOT_ESTABLISHED") { Process = "EXITED"; Exit = "ESTABLISHED"; return true; }
25|            InvalidSuffix = true; return false;
26|        }
27|    }

3|class PaperBranchTests {

10|        Check(s.Apply("EXECUTION_RELEASE_OBSERVED") && s.Process == "RELEASED" && s.Code == "NOT_ESTABLISHED", "resume not activity");
11|        Check(s.Apply("EXECUTION_ACTIVITY_OBSERVED") && s.Code == "ESTABLISHED", "activity");
12|        Check(s.Apply("OUTPUT_SEALED") && s.Output == "ESTABLISHED", "seal");
13|        Check(s.Apply("EXIT_OBSERVED") && s.Process == "EXITED" && s.Exit == "ESTABLISHED", "exit");
14|        var t = Created();
15|        Check(t.Apply("SUSPENDED_TERMINATION_INTENT") && t.Process == "SUSPENDED_TERMINATION_INTENT_UNCERTAIN", "termination intent");
16|        Check(t.Apply("SUSPENDED_TERMINATION_OBSERVED") && t.Process == "TERMINATED_SUSPENDED" && t.Code == "NON_EXECUTION_ESTABLISHED", "termination observed");
17|        Check(t.Apply("OUTPUT_SEALED") && t.Apply("EXIT_OBSERVED") && t.Code == "NON_EXECUTION_ESTABLISHED", "abort exit");

22|        noActivity.Apply("EXECUTION_RELEASE_OBSERVED");

25|        Check(noActivity.Apply("EXIT_OBSERVED") && noActivity.Exit == "ESTABLISHED"
26|            && noActivity.Code == "NOT_ESTABLISHED", "zero-activity exit retains uncertainty");
27|        Console.WriteLine("branch failures=" + failures); return failures == 0 ? 0 : 1;
28|    }

3|class PaperPrefixTests {
4|    static string Snapshot(PaperLifecycle s) { return s.Process+"/"+s.Code+"/"+s.Output+"/"+s.Exit; }
5|    static int Main() {
6|        string[][] branches = {
7|            new [] { "SPAWN_INTENT", "PROCESS_CREATED_SUSPENDED", "EXECUTION_RELEASE_INTENT", "EXECUTION_RELEASE_OBSERVED", "EXECUTION_ACTIVITY_OBSERVED", "OUTPUT_SEALED", "EXIT_OBSERVED" },
8|            new [] { "SPAWN_INTENT", "PROCESS_CREATED_SUSPENDED", "EXECUTION_RELEASE_INTENT", "EXECUTION_RELEASE_OBSERVED", "OUTPUT_SEALED", "EXIT_OBSERVED" },
9|            new [] { "SPAWN_INTENT", "PROCESS_CREATED_SUSPENDED", "SUSPENDED_TERMINATION_INTENT", "SUSPENDED_TERMINATION_OBSERVED", "OUTPUT_SEALED", "EXIT_OBSERVED" }
10|        };
11|        string[] events = { "SPAWN_INTENT", "PROCESS_CREATED_SUSPENDED", "EXECUTION_RELEASE_INTENT", "EXECUTION_RELEASE_OBSERVED", "EXECUTION_ACTIVITY_OBSERVED", "SUSPENDED_TERMINATION_INTENT", "SUSPENDED_TERMINATION_OBSERVED", "OUTPUT_SEALED", "EXIT_OBSERVED", "UNKNOWN", null };
12|        int checkedCases=0;
13|        foreach (var branch in branches) for (int prefix=0; prefix<=branch.Length; prefix++) foreach (var ev in events) {
14|            var s=new PaperLifecycle(); for(int i=0;i<prefix;i++) if(!s.Apply(branch[i])) throw new Exception("valid prefix rejected");
15|            string before=Snapshot(s);
16|            bool allowed=prefix<branch.Length && ev==branch[prefix];
17|            // At created-suspended either branch is allowed.
18|            if(prefix==2 && (ev=="EXECUTION_RELEASE_INTENT" || ev=="SUSPENDED_TERMINATION_INTENT")) allowed=true;
19|            // Validated zero-activity seal is allowed; only activity establishes code.
20|            if(prefix==4 && branch[2]=="EXECUTION_RELEASE_INTENT"
21|                && (ev=="EXECUTION_ACTIVITY_OBSERVED" || ev=="OUTPUT_SEALED")) allowed=true;
22|            if(s.Apply(ev)!=allowed) throw new Exception("wrong acceptance at "+prefix+" event "+ev);
23|            if(!allowed) {
24|                if(!s.InvalidSuffix || Snapshot(s)!=before) throw new Exception("invalid event changed prefix");
25|                foreach(var later in events) if(s.Apply(later) || Snapshot(s)!=before) throw new Exception("suffix recovered after failure");
26|            }
27|            checkedCases++;
28|        }
29|        Console.WriteLine("PASS prefix/event matrix cases="+checkedCases); return 0;
30|    }
