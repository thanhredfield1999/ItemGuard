15:

17:     private static final Set<SqliteProcessLock> RETAINED_UNTIL_EXIT =
18:         ConcurrentHashMap.newKeySet();
19: 
20:     private final Path lockPath;
21:     private final FileChannel channel;
22:     private final FileLock lock;
23: 
24:     private SqliteProcessLock(Path lockPath, FileChannel channel, FileLock lock) {
25:         this.lockPath = lockPath;
26:         this.channel = channel;
27:         this.lock = lock;
28:     }
29: 
30:     static SqliteProcessLock acquire(Path databasePath) {
31:         Path lockPath = deriveLockPath(databasePath);
32:         FileChannel channel = null;
33:         try {
34:             channel = FileChannel.open(
35:                 lockPath,
36:                 StandardOpenOption.CREATE,
37:                 StandardOpenOption.WRITE
38:             );
39:             FileLock lock = channel.tryLock();
40:             if (lock == null) {
41:                 closeQuietly(channel);
42:                 throw alreadyOwned(lockPath, null);
43:             }
44:             return new SqliteProcessLock(lockPath, channel, lock);
45:         } catch (OverlappingFileLockException overlapping) {
46:             closeQuietly(channel);
47:             throw alreadyOwned(lockPath, overlapping);
48:         } catch (IOException failure) {
49:             closeQuietly(channel);
50:             throw new IllegalStateException(
51:                 "Cannot acquire ItemGuard SQLite process lock: " + lockPath,
52:                 failure
53:             );
54:         }
55:     }
56: 
57:     private static Path deriveLockPath(Path databasePath) {
58:         Objects.requireNonNull(databasePath, "databasePath");
59:         Path absolute = databasePath.toAbsolutePath().normalize();
60:         Path parent = absolute.getParent();
61:         Path fileName = absolute.getFileName();
62:         if (parent == null || fileName == null) {
63:             throw new IllegalArgumentException("SQLite database path must have a parent and file name");
64:         }
65:         try {
66:             Files.createDirectories(parent);
67:             Path realParent = parent.toRealPath();
68:             return realParent.resolve(fileName + LOCK_SUFFIX);
69:         } catch (IOException failure) {
70:             throw new IllegalStateException(
71:                 "Cannot resolve ItemGuard SQLite database directory: " + parent,
72:                 failure
73:             );
74:         }
75:     }
76: 
77:     @Override
78:     public void close() {
79:         IOException failure = null;
80:         try {
81:             if (lock.isValid()) {
82:                 lock.release();
83:             }
84:         } catch (IOException releaseFailure) {
85:             failure = releaseFailure;
86:         }
87:         try {
88:             channel.close();
89:         } catch (IOException closeFailure) {
90:             if (failure == null) {
91:                 failure = closeFailure;
92:             } else {
93:                 failure.addSuppressed(closeFailure);
94:             }
95:         }
96:         if (failure != null) {
97:             throw new IllegalStateException(

99:                 failure
100:             );
101:         }
102:     }

106:     }

112:         );
113:     }
